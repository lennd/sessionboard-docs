import { Accordion } from '@compat';

<Accordion title="A video says no transcript yet">
A course is written from the transcript. Transcribe the video first, then draft the course again.
</Accordion>

<Accordion title="I can't publish a course">
**Approve and publish** stays off until every question is accepted, edited or rejected. The draft tells you how many still need a decision.
</Accordion>

<Accordion title="Credits is off for this organization">
An organization admin can select **Turn on Credits** on the **Credits** page.
</Accordion>

<Accordion title="A course from a meeting is held as a draft">
Sessions without a faculty disclosure are held back, and sessions without a recording wait for one. Add what is missing, then publish again.
</Accordion>

<Accordion title="I don't see Credits in the sidebar">
Credits only shows once it is on. Organizations set up for companies start with it off.
</Accordion>

<Accordion title="Learning isn't on in this organization yet">
An organization admin can select **Add Learning to this workspace** to start a 14-day trial.
</Accordion>

<Accordion title="My sign-in code didn't arrive">
Codes are good for 10 minutes. Check your spam folder, then request a new code.
</Accordion>
